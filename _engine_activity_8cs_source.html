310 public System.IO.Stream GetStreamFromUri(Uri uri, out string fileName) {
311 System.IO.Stream stream = null;
312 fileName = null;
313 try {
314 using (ICursor cursor = ContentResolver?.Query(uri, null, null, null, null)) {
315 if (cursor != null
316 && cursor.MoveToFirst()) {
317 int nameIndex = cursor.GetColumnIndex(IOpenableColumns.DisplayName);
318 if (nameIndex >= 0) {
319 fileName = cursor.GetString(nameIndex);
320 }
321 }
322 }
323 stream = ContentResolver?.OpenInputStream(uri);
324 }
325 catch {
326 // ignored
327 }
328 if (string.IsNullOrEmpty(fileName)) {
329 fileName = Path.GetFileName(uri.Path);
330 }
331 return stream;
